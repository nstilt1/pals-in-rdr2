cmake_minimum_required(VERSION 3.20)
project(RDR2PalPrototype VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_target(audit ALL COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_repository.py")
set(SCRIPTHOOK_SDK_ROOT "" CACHE PATH "External Script Hook SDK directory containing inc and lib")
if(NOT EXISTS "${SCRIPTHOOK_SDK_ROOT}/inc/main.h" OR NOT EXISTS "${SCRIPTHOOK_SDK_ROOT}/lib/ScriptHookRDR2.lib")
 message(FATAL_ERROR "Supply -DSCRIPTHOOK_SDK_ROOT=PATH to your separately obtained SDK; do not copy it into the repository.")
endif()
add_library(RDR2PalPrototype SHARED src/plugin.cpp)
target_compile_definitions(RDR2PalPrototype PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX RDR2PAL_VERSION="${PROJECT_VERSION}")
target_include_directories(RDR2PalPrototype PRIVATE "${SCRIPTHOOK_SDK_ROOT}/inc")
target_link_libraries(RDR2PalPrototype PRIVATE "${SCRIPTHOOK_SDK_ROOT}/lib/ScriptHookRDR2.lib" shell32)
set_target_properties(RDR2PalPrototype PROPERTIES PREFIX "" SUFFIX ".asi")
add_dependencies(RDR2PalPrototype audit)
enable_testing()
add_executable(targeting_tests tests/targeting_tests.cpp)
add_test(NAME targeting COMMAND targeting_tests)
add_executable(capture_tests tests/capture_tests.cpp)
add_test(NAME capture COMMAND capture_tests)
add_executable(companion_combat_tests tests/companion_combat_tests.cpp)
add_test(NAME companion_combat COMMAND companion_combat_tests)
add_executable(world_resources_tests tests/world_resources_tests.cpp)
add_test(NAME world_resources COMMAND world_resources_tests)
add_executable(party_tests tests/party_tests.cpp)
add_test(NAME party COMMAND party_tests)
add_test(NAME sphere_metadata COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/sphere_metadata_tests.py")
add_test(NAME pals_page COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/pals_page_tests.py")
set_tests_properties(sphere_metadata PROPERTIES ENVIRONMENT "RDR2PAL_TEST_TMP=${CMAKE_CURRENT_BINARY_DIR}")
set(CAPTURE_KIT_BRIDGE_DLL "" CACHE FILEPATH "Optional external Capture Kit Bridge DLL for C ABI tests")
if(CAPTURE_KIT_BRIDGE_DLL)
 add_executable(capture_bridge_tests tests/capture_bridge_tests.cpp)
 target_compile_definitions(capture_bridge_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_test(NAME capture_bridge COMMAND capture_bridge_tests "${CAPTURE_KIT_BRIDGE_DLL}")
endif()
add_executable(freeze_probe src/freeze_probe.cpp)
target_compile_definitions(freeze_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(freeze_probe PRIVATE dbghelp)
